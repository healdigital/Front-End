let done = new Set();
let activeFilter = 'all';

function setFilter(f) {
  activeFilter = f;
  document.querySelectorAll('.filter-btn').forEach(b => {
    b.classList.remove('active');
    if (b.textContent.startsWith(f === 'all' ? 'All' : f.split(' ')[0])) b.classList.add('active');
  });
  render();
}

function toggle(id) {
  if (done.has(id)) done.delete(id);
  else done.add(id);
  updateStats();
  render();
}

function updateStats() {
  const p1 = items.filter(i => i.priority === 'P1' && !done.has(i.id)).length;
  const p2 = items.filter(i => i.priority === 'P2' && !done.has(i.id)).length;
  document.getElementById('s-total').textContent = items.length;
  document.getElementById('s-p1').textContent = p1;
  document.getElementById('s-p2').textContent = p2;
  document.getElementById('s-done').textContent = done.size;
  document.getElementById('progress').style.width = Math.round(done.size / items.length * 100) + '%';
}

function render() {
  const filtered = items.filter(i => {
    if (activeFilter === 'all') return true;
    if (activeFilter === 'P1') return i.priority === 'P1';
    if (activeFilter === 'P2') return i.priority === 'P2';
    if (activeFilter === 'P3') return i.priority === 'P3';
    return i.cat === activeFilter;
  });

  const groups = {'P1': [], 'P2': [], 'P3': []};
  filtered.forEach(i => groups[i.priority].push(i));

  const labels = {'P1':'P1 — Critical (ship blockers)', 'P2':'P2 — Important (pre-launch)', 'P3':'P3 — Nice to have (post-launch)'};
  let html = '';

  ['P1','P2','P3'].forEach(p => {
    if (!groups[p].length) return;
    html += `<div class="section"><div class="section-title">${labels[p]}</div>`;
    groups[p].forEach(item => {
      const isDone = done.has(item.id);
      const tagClass = item.priority === 'P1' ? 'tag-p1' : item.priority === 'P2' ? 'tag-p2' : 'tag-p3';
      html += `<div class="item p${p[1]}${isDone ? ' done' : ''}">
        <div class="priority"></div>
        <div class="item-body">
          <div class="item-top">
            <span class="item-id">${item.id}</span>
            <span class="item-title">${item.title}</span>
          </div>
          <div class="item-meta">
            <span class="tag ${tagClass}">${item.priority}</span>
            <span class="tag tag-cat">${item.cat}</span>
            <span class="tag tag-time">${item.time}</span>
          </div>
          ${item.note ? `<div class="item-note">${item.note}</div>` : ''}
        </div>
        <div class="check${isDone ? ' checked' : ''}" onclick="toggle('${item.id}')">
          <svg width="10" height="10" viewBox="0 0 10 10" fill="none">
            <path d="M2 5l2.5 2.5L8 3" stroke="var(--color-text-secondary)" stroke-width="1.5" stroke-linecap="round" stroke-linejoin="round"/>
          </svg>
        </div>
      </div>`;
    });
    html += '</div>';
  });

  if (!html) html = '<div class="empty">No items match this filter.</div>';
  document.getElementById('list').innerHTML = html;
}

updateStats();
render();
